athlete_avg <- strava_runs |>
     group_by(athlete) |>
     summarise(
         avg_distance = mean(distance..m., na.rm = TRUE) / 1000,
     )
long_dis_runner <- athlete_avg |>
  filter(avg_distance > 13) |>
  mutate(runner_type = "Long-distance runner")
short_dis_runner <- athlete_avg |>
  filter(avg_distance < 7) |>
  mutate(runner_type = "Short-distance runner")
runner_groups <- full_join(
  long_dis_runner |> select(athlete, runner_type),
  short_dis_runner |> select(athlete, runner_type),
  by = "athlete"
)
runner_groups <- runner_groups |>
  mutate(
    runner_type = coalesce(runner_type.x, runner_type.y)
  ) |>
  select(athlete, runner_type)
data_7k_13k <- strava_runs |>
  left_join(
    runner_groups,
    by = "athlete"
  )
data_7k_13k <- data_7k_13k |>
  filter(
    !is.na(runner_type),
    !is.na(distance..m.),
    !is.na(elapsed.time..s.),
    elapsed.time..s. > 0
  ) |>
  mutate(
    distance_km = distance..m. / 1000,
    Run_average_speed = distance..m. / elapsed.time..s.
  )
ggplot(
  data_7k_13k,
  aes(
    x = distance_km,
    y = Run_average_speed,
    colour = runner_type
  )
) +
  geom_point(alpha = 0.2)+
  geom_smooth(method = "lm", se = FALSE) +
  scale_colour_manual(
    values = c(
      "Long-distance runner" = "orange",
      "Short-distance runner" = "forestgreen"
    )
  ) +
  labs(
    title = "Speed vs Distance for Long- and Short-Distance Runners",
    subtitle = "Groups based on average running distance: >13 km vs <7 km",
    x = "Run distance (km)",
    y = "Run average speed (m/s)",
  )
## `geom_smooth()` using formula = 'y ~ x'

# This graph suggests that long-distance runners generally maintain a relatively stable average speed as running distance increases, whereas short-distance runners show a more decrease in average speed. This shows that long-distance runners may be better able to maintain their pace over longer distances.
